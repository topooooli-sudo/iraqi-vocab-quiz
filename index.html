const SHEET_ID = "1JqjExw9fLtWglyYnxuT2w8PdIC8iYNdULTlXZBvKG8g"; function doPost(e) { try { const spreadsheet = SpreadsheetApp.openById(SHEET_ID); const sheet = spreadsheet.getSheets()[0]; const data = JSON.parse(e.postData.contents); const userId = String(data.userId || "").trim(); if (!userId) { return jsonResponse({ success: false, error: "شناسه ایتا دریافت نشد" }); } // بررسی اینکه این کاربر قبلاً آزمون داده یا نه const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const userIds = sheet.getRange(2, 1, lastRow - 1, 1) .getValues(); for (let i = 0; i < userIds.length; i++) { if (String(userIds[i][0]).trim() === userId) { return jsonResponse({ success: false, alreadyDone: true, message: "این کاربر قبلاً آزمون را انجام داده است" }); } } } // ثبت نتیجه جدید sheet.appendRow([ userId, data.score || "", data.percent || "", data.correct || "", data.wrong || "", new Date() ]); SpreadsheetApp.flush(); return jsonResponse({ success: true, message: "نتیجه با موفقیت ثبت شد", row: sheet.getLastRow() }); } catch (error) { return jsonResponse({ success: false, error: error.toString() }); } } function doGet(e) { try { const spreadsheet = SpreadsheetApp.openById(SHEET_ID); const sheet = spreadsheet.getSheets()[0]; const userId = e && e.parameter && e.parameter.userId ? String(e.parameter.userId).trim() : ""; let alreadyDone = false; if (userId && sheet.getLastRow() >= 2) { const userIds = sheet .getRange(2, 1, sheet.getLastRow() - 1, 1) .getValues(); for (let i = 0; i < userIds.length; i++) { if (String(userIds[i][0]).trim() === userId) { alreadyDone = true; break; } } } return jsonResponse({ connected: true, alreadyDone: alreadyDone, spreadsheet: spreadsheet.getName(), sheet: sheet.getName(), rows: sheet.getLastRow() }); } catch (error) { return jsonResponse({ connected: false, error: error.toString() }); } } function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); }
